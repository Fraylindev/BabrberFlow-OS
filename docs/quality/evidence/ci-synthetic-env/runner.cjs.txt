function sanitize(t){for(const m of t.matchAll(/::add-mask::([^\r\n]+)/g))if(!secrets.includes(m[1]))secrets.push(m[1]);for(const s of secrets)t=t.replaceAll(s,'[omitted]');return t.replace(/eyJ[A-Za-z0-9_-]+\.[A-Za-z0-9_-]+\.[A-Za-z0-9_-]+/g,'[JWT omitted]').replace(/postgres(?:ql)?:\/\/[^\s"']+/gi,'[database URL omitted]');}
function run(command,args,label,env=sys,options={}){const r=spawnSync(command,args,{cwd:clean,env,windowsHide:true,encoding:'utf8',timeout:900000,maxBuffer:30e6,...options});const log=sanitize((r.stdout||'')+(r.stderr||'')+(r.error?.message||''));fs.writeFileSync(path.join(output,label+'.log'),log);report.commands.push({label,exitCode:r.status});console.log(label+': '+r.status);if(r.status!==0)throw Error(label+' failed; see sanitized log');return log;}
