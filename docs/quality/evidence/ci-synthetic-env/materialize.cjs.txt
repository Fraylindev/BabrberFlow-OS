const fs=require('fs'),path=require('path');const root=process.cwd(),clean=path.join(root,'.tmp/ci-synthetic-clean');
const source=path.join(root,'node_modules/.pnpm'),dest=path.join(clean,'node_modules/.pnpm');
if(path.dirname(clean)!==path.join(root,'.tmp'))throw Error('Own checkout boundary');
if(fs.lstatSync(dest).isSymbolicLink())fs.unlinkSync(dest);fs.mkdirSync(dest,{recursive:true});let files=0,links=0;
function mapped(real){if(real.toLowerCase().startsWith(dest.toLowerCase()+path.sep))return real;if(real.toLowerCase().startsWith(source.toLowerCase()+path.sep))return path.join(dest,path.relative(source,real));if([path.join(root,'apps/api'),path.join(root,'apps/web')].includes(real))return path.join(clean,path.relative(root,real));throw Error('Package target outside installed store: '+real);}
function copy(from,to){for(const e of fs.readdirSync(from,{withFileTypes:true})){const a=path.join(from,e.name),b=path.join(to,e.name);if(e.isSymbolicLink()){try{fs.lstatSync(b);continue;}catch{}const target=mapped(fs.realpathSync(a));fs.symlinkSync(target,b,fs.statSync(a).isDirectory()?'junction':'file');links++;}else if(e.isDirectory()){fs.mkdirSync(b,{recursive:true});copy(a,b);}else if(e.isFile()){if(fs.existsSync(b))continue;try{fs.linkSync(a,b);}catch{fs.copyFileSync(a,b);}files++;}}}
copy(source,dest);
for(const rel of ['node_modules','apps/api/node_modules','apps/web/node_modules']){const d=path.join(clean,rel);function repair(p){for(const e of fs.readdirSync(p,{withFileTypes:true})){const a=path.join(p,e.name);if(e.name==='.pnpm')continue;if(e.isSymbolicLink()){if(['.cache','.vite-temp','.vite'].includes(e.name)){fs.unlinkSync(a);fs.mkdirSync(a);continue;}if(e.name==='.bin'){fs.unlinkSync(a);fs.mkdirSync(a);const orig=path.join(root,rel,'.bin');for(const f of fs.readdirSync(orig)){const from=path.join(orig,f);if(fs.statSync(from).isFile()){let t=fs.readFileSync(from,'utf8');t=t.replaceAll(clean.replaceAll('\\','/'),root.replaceAll('\\','/')).replaceAll(clean,root);t=t.replaceAll(root.replaceAll('\\','/'),clean.replaceAll('\\','/')).replaceAll(root,clean);fs.writeFileSync(path.join(a,f),t);}}}else{const target=mapped(fs.realpathSync(a));fs.unlinkSync(a);fs.symlinkSync(target,a,'junction');}}else if(e.isDirectory())repair(a);}}repair(d);}
console.log(JSON.stringify({hardlinkedFiles:files,packageLinks:links,versionsChanged:false}));
