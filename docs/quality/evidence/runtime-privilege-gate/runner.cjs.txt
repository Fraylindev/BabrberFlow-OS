function clean(t){for(const s of Object.values(secrets))t=t.replaceAll(s,'[omitted]');return t.replace(/^::add-mask::.*$/gm,'::add-mask::[omitted]').replace(/postgres(?:ql)?:\/\/[^\s"']+/gi,'[local URL omitted]');}
function run(cmd,args,label,opts={}){
 const r=spawnSync(cmd,args,{cwd:root,env,encoding:'utf8',windowsHide:true,timeout:120000,maxBuffer:8e6,...opts});
 const log=clean((r.stdout||'')+(r.stderr||'')+(r.error?.message||''));
 fs.writeFileSync(path.join(output,label+'.log'),log);
 report.commands.push({label,exitCode:r.status});console.log(label+': '+r.status);
 if(r.status!==0&&!opts.allowed)throw Error(label+' failed');return {exitCode:r.status,log};
}
function sql(label,user,db,text,source,allowed=false){return run(path.join(bin,'psql.exe'),['-X','-h','127.0.0.1','-p',String(port),'-U',user,'-d',db,'-v','ON_ERROR_STOP=1',...(source?['-f',source]:['-At'])],label,{input:text,allowed,env:{...env,PGPASSWORD:secrets[user==='postgres'?'admin':user==='kortek_ci_runner'?'runner':user==='kortek_migrator'?'migrator':'runtime']}});}

function migrate(base,db,user,label){run(process.execPath,['--require',preload,prisma,'migrate','deploy','--schema',path.join(base,'apps/api/prisma/schema.prisma')],label,{cwd:path.join(base,'apps/api'),env:{...env,NODE_OPTIONS:`--require="${preload.replaceAll('\\','/')}"`,DATABASE_URL:`postgresql://${user}:${secrets[user==='kortek_ci_runner'?'runner':'migrator']}@127.0.0.1:${port}/${db}?schema=public`}});}
