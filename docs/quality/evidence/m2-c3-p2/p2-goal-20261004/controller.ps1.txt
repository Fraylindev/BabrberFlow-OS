function Send-PrivateLine([Security.SecureString] $secret, [IO.StreamWriter] $writer) {
    $localPointer = [IntPtr]::Zero
    $temporaryValue = $null
    try {
        $localPointer = [Runtime.InteropServices.Marshal]::SecureStringToBSTR($secret)
        $temporaryValue = [Runtime.InteropServices.Marshal]::PtrToStringBSTR($localPointer)
        $writer.WriteLine($temporaryValue)
        $writer.Flush()
    } finally {
        $temporaryValue = $null
        if ($localPointer -ne [IntPtr]::Zero) { [Runtime.InteropServices.Marshal]::ZeroFreeBSTR($localPointer) }
    }
}
function Compare-Private([Security.SecureString] $first, [Security.SecureString] $second) {
    if ($first.Length -ne $second.Length -or $first.Length -eq 0) { return $false }
    $firstPtr = [IntPtr]::Zero
    $secondPtr = [IntPtr]::Zero
    try {
        $firstPtr = [Runtime.InteropServices.Marshal]::SecureStringToBSTR($first)
        $secondPtr = [Runtime.InteropServices.Marshal]::SecureStringToBSTR($second)
        $difference = 0
        for ($i = 0; $i -lt $first.Length; $i++) {
            $difference = $difference -bor ([Runtime.InteropServices.Marshal]::ReadInt16($firstPtr, 2 * $i) -bxor [Runtime.InteropServices.Marshal]::ReadInt16($secondPtr, 2 * $i))
        }
        return $difference -eq 0
    } finally {
        if ($firstPtr -ne [IntPtr]::Zero) { [Runtime.InteropServices.Marshal]::ZeroFreeBSTR($firstPtr) }
        if ($secondPtr -ne [IntPtr]::Zero) { [Runtime.InteropServices.Marshal]::ZeroFreeBSTR($secondPtr) }
    }
}

function Save-SupervisorFailure($errorRecord, [string] $phase) {
    $message = ($errorRecord.Exception.Message -split "`r?`n")[0]
    $message = $message -replace '(?i)(postgres(?:ql)?|https?|ssh)://\S+', '[direccion omitida]'
    $message = $message -replace '"[^"\r\n]*"', '[literal omitido]'
    $message = $message -replace "'[^'\r\n]*'", '[literal omitido]'
    $message = $message -replace '(?i)(password|passphrase|token|secret|clave)\s*[:=]\s*\S+', '$1=[privado]'
    $failure = [ordered]@{ event='fallo-supervisor'; phase=$phase; exceptionClass=$errorRecord.Exception.GetType().Name;
        messageSanitized=$message; tracebackSanitized=@([ordered]@{
            file=[IO.Path]::GetFileName($errorRecord.InvocationInfo.ScriptName);
            line=$errorRecord.InvocationInfo.ScriptLineNumber; function='supervisor' }) }
    $json = $failure | ConvertTo-Json -Depth 5
    [IO.File]::WriteAllText((Join-Path $PSScriptRoot 'supervisor-error.json'), $json, [Text.UTF8Encoding]::new($false))
    Write-Host $json
}
