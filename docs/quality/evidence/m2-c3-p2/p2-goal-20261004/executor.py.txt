def emit(event, **fields):

    REPORT['event']=event

def sanitize_message(value):
    # No payloads, SQL DETAIL/CONTEXT, literals, credential values or command argv.
    text=str(value)
    for secret in PRIVATE_VALUES:
        if secret: text=text.replace(secret,'[privado]')
    if PG_ENV and PG_ENV.get('PGPASSWORD'): text=text.replace(PG_ENV['PGPASSWORD'],'[privado]')
    for secret in [PHRASE,DECRYPT_PHRASE]:
        if secret: text=text.replace(bytes(secret).decode('utf-8',errors='replace'),'[privado]')
    text=re.sub(r'(?:postgres(?:ql)?|https?|ssh)://[^\s]+','[direccion omitida]',text,flags=re.I)
    text=re.sub(r'(?i)(password|passphrase|token|secret|clave)\s*[:=]\s*[^\s,;]+',r'\1=[privado]',text)
    text=re.sub(r'"[^"\r\n]*"','[literal omitido]',text)
    text=re.sub("'[^'\r\n]*'",'[literal omitido]',text)
    text=re.sub(r'(?i)(Failing row contains).*',r'\1 [fila omitida]',text)
    text=re.sub(r'(?i)(DETAIL:.*Key.*?)=\([^\r\n]*\)',r'\1=([valor omitido])',text)
    return text

def failure_info(error, phase=None, event='parada'):
    frames=[{'file':Path(f.filename).name,'line':f.lineno,'function':f.name}
            for f in traceback.extract_tb(error.__traceback__)]
    message=sanitize_message(error).encode('utf-8')[:8192].decode('utf-8',errors='ignore')
    if isinstance(error,subprocess.TimeoutExpired): message='comando supero el timeout: '+str(error.timeout)+' s'
    if isinstance(error,subprocess.CalledProcessError): message='comando devolvio exit '+str(error.returncode)
    return {'event':event,'phase':phase or STAGE,'exceptionClass':type(error).__name__,
            'messageSanitized':message,'queryLabel':LAST_QUERY,'tracebackSanitized':frames}

def retain_failure(error, phase=None, event='parada'):
    item=failure_info(error,phase,event)
    REPORT['success']=False
    REPORT.setdefault('errors',[]).append(item)
    REPORT.setdefault('failure',item)
    try: save()
    except Exception as handler_error:
        REPORT['errors'].append(failure_info(handler_error,'persistencia-error','fallo-manejador'))
        # Fallback separado: no depende de save ni del emisor fallido.
        (WORK/'failure-fallback.json').write_text(json.dumps(REPORT,indent=2,ensure_ascii=True)+'\n',encoding='utf-8')
    return item

def report_failure(error):
    item=retain_failure(error)
    try: emit('parada',failure=item)
    except Exception as handler_error: retain_failure(handler_error,'manejador-error','fallo-manejador')

def safe_stderr(data):
    return sanitize_message(data.decode('utf-8',errors='replace')).encode('utf-8')[:8192].decode('utf-8',errors='ignore')

def command_record(args, code, stderr, label=None, error=None):
    redacted=sanitize_message(stderr.decode('utf-8',errors='replace')).encode('utf-8')
    safe=['--snapshot=<snapshot-en-memoria>' if x.startswith('--snapshot=') else sanitize_message(x) for x in args]
    frames=[]
    if error is not None:
        frames=failure_info(error)['tracebackSanitized']
    elif code:
        frames=[{'file':Path(f.filename).name,'line':f.lineno,'function':f.name}
                for f in traceback.extract_stack()[:-1]]
    item={'args':safe,'commandSanitized':safe,'exitCode':code,'label':label,'phase':STAGE,
          'stderrCompleteRedacted':redacted[:8192].decode('utf-8',errors='ignore'),
          'stderrTruncated':len(redacted)>8192,'stderrCapBytes':8192,
          'exceptionClass':type(error).__name__ if error is not None else 'ExternalCommandExit' if code else None,
          'tracebackSanitized':frames}
    REPORT['commands'].append(item);save()
    return item

def command_failure(label, code, stderr):
    REPORT['failureCommand']={'label':label,'exitCode':code,'messageSanitized':safe_stderr(stderr)}
    raise RuntimeError(str(label or 'comando-fallido')+'; exit='+str(code)+'; '+safe_stderr(stderr))

def spawn_pipe(args, **kwargs):
    try: process=subprocess.Popen(args,cwd=ROOT,stderr=subprocess.PIPE,**kwargs)
    except BaseException as error:
        command_record(args,None,b'',label='crear-proceso-pipe',error=error)
        raise
    errors=[]
    def drain():
        while True:
            part=process.stderr.read(8192)
            if not part: break
            errors.append(part)
    worker=threading.Thread(target=drain,daemon=True); worker.start()
    process.p2_errors=errors; process.p2_error_worker=worker
    process.p2_args=args;process.p2_recorded=False
    return process

def pipe_error(process):
    process.p2_error_worker.join(timeout=5)
    return b''.join(process.p2_errors)

def record_pipe(process,label):
    if not process.p2_recorded:
        command_record(process.p2_args,process.returncode,pipe_error(process),label)
        process.p2_recorded=True

def check(label, **fields):

def run(args, data=None, env=None, timeout=120, expected=0, label=None, record=True):
    try:
        process=subprocess.Popen(args,cwd=ROOT,stdin=subprocess.PIPE if data is not None else subprocess.DEVNULL,
                                 stdout=subprocess.PIPE,stderr=subprocess.PIPE,env=env or BASE_ENV)
    except BaseException as error:
        command_record(args,None,b'',label,error=error)
        raise
    try:
        output,stderr=process.communicate(input=data,timeout=timeout)
    except BaseException as error:
        if process.poll() is None: process.kill()
        output,stderr=process.communicate(timeout=15)
        command_record(args,process.returncode,stderr,label,error=error)
        raise
    command_record(args,process.returncode,stderr,label)
    if expected is not None and process.returncode != expected:
        command_failure(label,process.returncode,stderr)
    return output

class Reader:
    def __init__(self, client):
        keys=['PGPASSWORD','PGHOST','PGPORT','PGUSER','PGDATABASE','PGSSLMODE','PGSSLROOTCERT','PGOPTIONS','PGAPPNAME','PGCLIENTENCODING']
        opts=sum((['-e',k] for k in keys),[])
        args=DOCKER+['exec','-i']+opts+[client,'psql','-X','-qAt','-v','ON_ERROR_STOP=1']
        try:
            self.proc=subprocess.Popen(args,cwd=ROOT,env=PG_ENV,stdin=subprocess.PIPE,
                                       stdout=subprocess.PIPE,stderr=subprocess.PIPE,text=True,encoding='utf-8',errors='strict')
        except BaseException as error:
            command_record(args,None,b'',label='crear-reader',error=error)
            raise
        self.lines=queue.Queue()
        self.stderr=[]
        self.args=args;self.recorded=False
        def error_pump():
            for line in self.proc.stderr:
                self.stderr.append(line)
        self.error_thread=threading.Thread(target=error_pump,daemon=True);self.error_thread.start()
        def pump():
            for line in self.proc.stdout: self.lines.put(line.rstrip('\r\n'))
            self.lines.put(None)
        threading.Thread(target=pump,daemon=True).start()
        try: self.q("BEGIN TRANSACTION ISOLATION LEVEL REPEATABLE READ READ ONLY; SET LOCAL statement_timeout='120s'")
        except BaseException:
            self.kill()
            raise
    def q(self, sql, label='consulta-scalar'):
        global LAST_QUERY
        LAST_QUERY=label
        end='P2_END_'+secrets.token_hex(8)
        self.proc.stdin.write(sql+";\nSELECT '"+end+"';\n"); self.proc.stdin.flush()
        rows=[]
        while True:
            line=self.lines.get(timeout=150)
            if line is None:
                self.proc.wait(timeout=15);self.record()
                raise RuntimeError('reader-fuente-fallo; '+safe_stderr(''.join(self.stderr).encode('utf-8')))
            if line==end: return rows
            if line: rows.append(line)
    def json(self, sql, label=None):
        label=label or {LEDGER_SQL:'ledger-tecnico',CATALOG_SQL:'catalogo',ROLE_SQL:'roles',COUNTS_SQL:'conteos'}.get(sql,'login-readonly')
        return json.loads('\n'.join(self.q(sql,label=label)))
    def close(self):
        if self.proc.poll() is None:
            self.q('COMMIT'); self.proc.stdin.close()
            self.proc.wait(timeout=15)
        self.record()
        if self.proc.returncode!=0: raise RuntimeError('reader-fuente-exit; '+safe_stderr(''.join(self.stderr).encode('utf-8')))
    def kill(self):
        if self.proc.poll() is None: self.proc.kill(); self.proc.wait(timeout=15)
        self.record()
    def record(self):
        if not self.recorded:
            self.error_thread.join(timeout=5)
            command_record(self.args,self.proc.returncode,''.join(self.stderr).encode('utf-8'),'reader-fuente-RR-RO')
            self.recorded=True

def request_phrase(kind):
    global DECRYPT_PHRASE
    emit('prompt-frase',kind=kind)
    line=sys.stdin.buffer.readline()
    if not line: raise RuntimeError('prompt-cerrado')
    value=bytearray(line.rstrip(b'\r\n')); line=None
    if not value: raise RuntimeError('frase-vacia')
    if kind=='cifrar-doble-confirmacion':
        second_line=sys.stdin.buffer.readline()
        second=bytearray(second_line.rstrip(b'\r\n')); second_line=None
        same=bool(second) and secrets.compare_digest(value,second)
        DECRYPT_PHRASE=second if same else None
        if not same: erase(second)
        second=None
        if not same:
            erase(value)
            raise RuntimeError('frases-no-coinciden-antes-dump')
    PRIVATE_VALUES.append(bytes(value).decode('utf-8'))
    return value

def gpg_args(phrase, path, encrypt=True):
    global GPG_HOME
    if GPG_HOME is None:
        home=ROOT/'.tmp'/('kh-'+RUN[:6])
        if ROOT not in home.resolve().parents or home.exists(): raise RuntimeError('homedir-GPG-no-propio')
        if len(msys(home)+'/S.gpg-agent')>90: raise RuntimeError('socket-GPG-supera90')
        home.mkdir();GPG_HOME=home
        REPORT['ownGpgHome']={'path':str(home),'socketCharacters':len(msys(home)+'/S.gpg-agent')}
    home=GPG_HOME
    args=[GPG,'--no-options','--homedir',msys(home),'--no-symkey-cache','--batch',
          '--pinentry-mode','loopback','--passphrase-fd','0']
    return args+(['--symmetric','--cipher-algo','AES256','--output',msys(path)] if encrypt else ['--decrypt',msys(path)])

def encrypt_data(phrase, data, dest):
    # GPG consume la primera línea del stdin como frase; el resto es el payload.
    temporary=dest.with_suffix(dest.suffix+'.partial')
    try:
        run(gpg_args(phrase,temporary),data=bytes(phrase)+b'\n'+data,label='cifrado-AES256-stdin-anonimo')
        temporary.replace(dest)
    finally:
        if temporary.exists(): temporary.unlink()

def decrypt_data(phrase, path):
    return run(gpg_args(phrase,path,False),data=bytes(phrase)+b'\n',label='descifrado-stdin-anonimo')

def canonical_ledger(rows):
    # Multiconjunto técnico: conserva cada entrada, checksum y estado histórico.
    return sorted((r['name'],r['checksum'],r['finished'],r['rolledBack']) for r in rows)

def ledger_equal(left, right):
    return canonical_ledger(left)==canonical_ledger(right)

def observe_ledger(phase, rows):
    from collections import Counter
    baseline=baseline_ledger()
    actual=Counter(canonical_ledger(rows)); expected=Counter(canonical_ledger(baseline))
    def differences(values):
        return [dict(zip(['name','checksum','finished','rolledBack'],key),multiplicity=count)
                for key,count in sorted(values.items())]
    delta={'missing':differences(expected-actual),'extra':differences(actual-expected)}
    REPORT.setdefault('ledgerComparisons',[]).append({'phase':phase,'observed':rows,
        'baseline':baseline,'delta':delta,'equal':not delta['missing'] and not delta['extra']})
    save()
    if delta['missing'] or delta['extra']: raise RuntimeError('ledger-difiere-P0')

def roles_prepare(container, catalog, roles, database='postgres'):
    if roles['memberships']: raise RuntimeError('membresias-de-interes-no-vacias')
    owners={x['owner'] for kind in ['tables','types','functions'] for x in catalog[kind]}
    owners.add(catalog['schemaOwner'])
    attrs={x['rolname']:x for x in roles['attributes']}
    for name in sorted(owners|set(attrs)):
        if name=='postgres': continue
        if name=='pg_database_owner':
            exists=local_sql(container,"SELECT EXISTS(SELECT 1 FROM pg_roles WHERE rolname='pg_database_owner');",database)
            if exists!='t':raise RuntimeError('rol-incorporado-pg_database_owner-ausente')
            continue
        flags=attrs.get(name)
        if flags:
            options=' '.join(yes if flags[key] else no for key,yes,no in [
                ('rolcanlogin','LOGIN','NOLOGIN'),('rolinherit','INHERIT','NOINHERIT'),
                ('rolsuper','SUPERUSER','NOSUPERUSER'),('rolcreatedb','CREATEDB','NOCREATEDB'),
                ('rolcreaterole','CREATEROLE','NOCREATEROLE'),('rolreplication','REPLICATION','NOREPLICATION'),
                ('rolbypassrls','BYPASSRLS','NOBYPASSRLS')])
        else: options='NOLOGIN NOSUPERUSER NOCREATEDB NOCREATEROLE NOREPLICATION NOBYPASSRLS NOINHERIT'
        local_sql(container,'CREATE ROLE '+ident(name)+' '+options+';',database)

def roles_apply(container, catalog, roles, database='postgres'):
    commands=['ALTER SCHEMA public OWNER TO '+ident(catalog['schemaOwner'])+';',
              'REVOKE ALL ON SCHEMA public FROM PUBLIC,kortek_runtime,kortek_migrator,kortek_backup;',
              'REVOKE ALL ON ALL TABLES IN SCHEMA public FROM PUBLIC,kortek_runtime,kortek_migrator,kortek_backup;']
    for table in catalog['tables']:
        commands.append('ALTER TABLE public.'+ident(table['name'])+' OWNER TO '+ident(table['owner'])+';')
    for t in catalog['types']:
        commands.append('ALTER TYPE public.'+ident(t['name'])+' OWNER TO '+ident(t['owner'])+';')
    for f in catalog['functions']:
        commands+=['ALTER FUNCTION '+f['signature']+' OWNER TO '+ident(f['owner'])+';',
                   'REVOKE ALL ON FUNCTION '+f['signature']+' FROM PUBLIC,kortek_runtime,kortek_migrator,kortek_backup;']
    for p in roles['tablePrivileges']:
        if p['allowed']: commands.append('GRANT '+p['privilege']+' ON TABLE public.'+ident(p['table_name'])+' TO '+ident(p['role'])+';')
    for p in roles['schemaPrivileges']:
        if p['allowed']: commands.append('GRANT '+p['privilege']+' ON SCHEMA public TO '+ident(p['role'])+';')
    for p in roles['functionPrivileges']:
        if p['allowed']: commands.append('GRANT EXECUTE ON FUNCTION '+p['signature']+' TO '+ident(p['role'])+';')
    kinds={'r':'TABLES','S':'SEQUENCES','f':'FUNCTIONS','T':'TYPES','n':'SCHEMAS'}
    for p in roles['defaults']:
        prefix='ALTER DEFAULT PRIVILEGES FOR ROLE '+ident(p['owner'])
        if p['namespace']: prefix+=' IN SCHEMA '+ident(p['namespace'])
        target='PUBLIC' if p['grantee']=='PUBLIC' else ident(p['grantee'])
        commands.append(prefix+' GRANT '+p['privilege']+' ON '+kinds[p['kind']]+' TO '+target+(' WITH GRANT OPTION' if p['grantable'] else '')+';')
    local_sql(container,'BEGIN;\n'+'\n'.join(commands)+'\nCOMMIT;',database)

def filter_toc(toc):
    selected=[]; removed=[]
    for line in toc.splitlines():
        if line.startswith(';') or not line.strip(): selected.append(line); continue
        if re.search(r'\bSCHEMA - public\b|\bEXTENSION - btree_gist\b|\bCOMMENT - EXTENSION btree_gist\b|\bCOMMENT - SCHEMA public\b',line):
            removed.append(line)
        else: selected.append(line)
    if any('TABLE DATA' in x for x in removed): raise RuntimeError('TOC-omite-tablas')
    REPORT['tocBootstrapEntriesExcluded']=len(removed)
    return ('\n'.join(selected)+'\n').encode('utf-8')

def stream_encrypt(client, snapshot, phrase, dest):
    args=dump_args(client,snapshot)
    temporary=dest.with_suffix(dest.suffix+'.partial')
    src=enc=None
    try:
        src=spawn_pipe(args,env=PG_ENV,stdout=subprocess.PIPE)
        enc_args=gpg_args(phrase,temporary)
        enc=spawn_pipe(enc_args,env=BASE_ENV,stdin=subprocess.PIPE,stdout=subprocess.DEVNULL)
        enc.stdin.write(bytes(phrase)+b'\n'); enc.stdin.flush()
        shutil.copyfileobj(src.stdout,enc.stdin,1024*1024)
        enc.stdin.close(); src.stdout.close()
        src_code=src.wait(timeout=150); enc_code=enc.wait(timeout=150)
        record_pipe(src,'pg-dump-snapshot');record_pipe(enc,'gpg-cifrar-dump')
        if src_code: command_failure('pg-dump-snapshot',src_code,pipe_error(src))
        if enc_code: command_failure('gpg-cifrar-dump',enc_code,pipe_error(enc))
        temporary.replace(dest)
    finally:
        for process in [src,enc]:
            if process and process.poll() is None: process.kill(); process.wait(timeout=15)
            if process: record_pipe(process,'pipe-dump-cifrado-finalizacion')
        if temporary.exists(): temporary.unlink()

def stream_decrypt_restore(container, phrase, path, database='postgres'):
    args=gpg_args(phrase,path,False)
    dec=spawn_pipe(args,env=BASE_ENV,stdin=subprocess.PIPE,stdout=subprocess.PIPE)
    restore_args=DOCKER+['exec','-i','-e','PGCLIENTENCODING=UTF8',container,'pg_restore','-U','postgres','-d',database,
                        '--single-transaction','--exit-on-error','--no-owner','--no-privileges','--use-list=/tmp/p2-restore.list']
    proc=spawn_pipe(restore_args,env=BASE_ENV,stdin=dec.stdout,stdout=subprocess.DEVNULL)
    dec.stdout.close()
    try:
        dec.stdin.write(bytes(phrase)+b'\n'); dec.stdin.close()
        b=proc.wait(timeout=300); a=dec.wait(timeout=30)
        record_pipe(dec,'gpg-descifrar-restore');record_pipe(proc,'pg-restore')
        if a: command_failure('gpg-descifrar-restore',a,pipe_error(dec))
        if b: command_failure('pg-restore',b,pipe_error(proc))
    finally:
        for p in [dec,proc]:
            if p.poll() is None: p.kill(); p.wait(timeout=15)
            record_pipe(p,'pipe-descifrado-restore-finalizacion')

def decrypt_toc(container, phrase, path):
    args=gpg_args(phrase,path,False)
    dec=spawn_pipe(args,env=BASE_ENV,stdin=subprocess.PIPE,stdout=subprocess.PIPE)
    proc=spawn_pipe(DOCKER+['exec','-i',container,'pg_restore','--list'],env=BASE_ENV,stdin=dec.stdout,stdout=subprocess.PIPE)
    dec.stdout.close()
    try:
        dec.stdin.write(bytes(phrase)+b'\n'); dec.stdin.close()
        # La lectura de stderr pertenece exclusivamente al drenador de spawn_pipe.
        listing=proc.stdout.read();proc.wait(timeout=180);a=dec.wait(timeout=30)
        record_pipe(dec,'gpg-descifrar-TOC');record_pipe(proc,'pg-restore-list')
        if a: command_failure('gpg-descifrar-TOC',a,pipe_error(dec))
        if proc.returncode: command_failure('pg-restore-list',proc.returncode,pipe_error(proc))
        docker('exec','-i',container,'sh','-c','cat > /tmp/p2-restore.list',
               data=filter_toc(listing.decode('utf-8')),label='TOC-public-btree-tmpfs')
    finally:
        for p in [dec,proc]:
            if p.poll() is None: p.kill(); p.wait(timeout=15)
            record_pipe(p,'pipe-TOC-finalizacion')


def normalized_schema(text):
    # Tokens de encuadre aleatorios de pg_dump 17.11; no son SQL ni objetos.
    return '\n'.join(line.rstrip() for line in text.replace('\r\n','\n').splitlines()
      if not re.match(r'^\\(?:un)?restrict\s+\S+$',line)
      and not re.match(r'^-- Dumped (?:from database|by pg_dump) version ',line))+'\n'

def schema_categories(text):
    categories={k:[] for k in ['tables','columns','constraints','indexes','triggers','functions','sequences','extensions','types','other']}
    chunks=re.split(r'(?m)(?=^-- Name: .*; Type: .*; Schema:)',text)
    for chunk in chunks:
        match=re.search(r'^-- Name: (.*); Type: (.*); Schema:',chunk,re.M)
        if not match:
            categories['other'].append(chunk);continue
        typ=match[2]
        category=('tables' if typ=='TABLE' else 'constraints' if 'CONSTRAINT' in typ else
          'indexes' if 'INDEX' in typ else 'triggers' if 'TRIGGER' in typ else
          'functions' if typ in ['FUNCTION','PROCEDURE'] else 'sequences' if 'SEQUENCE' in typ else
          'extensions' if typ=='EXTENSION' or (typ=='COMMENT' and match[1].startswith('EXTENSION ')) else
          'types' if typ in ['TYPE','DOMAIN'] else 'other')
        categories[category].append(chunk)
        if category=='tables': categories['columns'].append(chunk)
    return categories

def schema_snapshot(container,snapshot,database='postgres'):
    args=dump_args(container,snapshot,'--schema-only','--format=plain')
    text=run(args,env=PG_ENV,label='schema-normalizado-mismo-snapshot').decode('utf-8')
    return normalized_schema(text)

def local_schema(container,database='postgres'):
    args=DOCKER+['exec','-e','PGCLIENTENCODING=UTF8',container,'pg_dump','-U','postgres','-d',database,
      '--schema=public','--schema-only','--no-owner','--no-privileges','--encoding=UTF8','--format=plain']
    return normalized_schema(run(args,label='schema-normalizado-restore').decode('utf-8'))

def strip_outer(value):
    value=value.strip()
    while value.startswith('(') and value.endswith(')'):
        depth=0;quote=None;encloses=True;i=0
        while i<len(value):
            c=value[i]
            if quote:
                if c==quote:
                    if i+1<len(value) and value[i+1]==quote: i+=1
                    else: quote=None
            elif c in ["'",'"']: quote=c
            elif c=='(': depth+=1
            elif c==')':
                depth-=1
                if depth==0 and i!=len(value)-1: encloses=False;break
            i+=1
        if not encloses or depth: break
        value=value[1:-1].strip()
    return value

def and_terms(value):
    value=strip_outer(value);depth=0;quote=None;parts=[];begin=0;i=0
    while i<len(value):
        c=value[i]
        if quote:
            if c==quote:
                if i+1<len(value) and value[i+1]==quote:i+=1
                else:quote=None
        elif c in ["'",'"']:quote=c
        elif c=='(':depth+=1
        elif c==')':depth-=1
        elif depth==0 and value[i:i+5]==' AND ':
            parts.append(value[begin:i]);begin=i+5;i+=4
        i+=1
    if not parts:return [value]
    parts.append(value[begin:])
    return [term for part in parts for term in and_terms(part)]

def check_lines(text):
    found={}
    for line in text.splitlines():
        match=re.fullmatch(r'    CONSTRAINT "([^"\n]+)" CHECK (.+?)(,?)',line)
        if match:found[match[1]]=(line,match[2],match[3])
    return found

def schema_compare(snapshot,restored,label,container=None):
    import difflib
    raw_snapshot=snapshot;raw_restore=restored
    lchecks=check_lines(snapshot);rchecks=check_lines(restored);accepted=[]
    for name in lchecks.keys() & rchecks.keys():
        left=lchecks[name];right=rchecks[name]
        if left==right:continue
        if left[2]!=right[2] or and_terms(left[1])!=and_terms(right[1]):continue
        if container is None or not check_truth(container,name,left[1],right[1]):continue
        canonical='    CONSTRAINT '+ident(name)+' CHECK ('+' AND '.join('('+x+')' for x in and_terms(left[1]))+')'+left[2]
        snapshot=snapshot.replace(left[0],canonical);restored=restored.replace(right[0],canonical);accepted.append(name)
    left=schema_categories(snapshot);right=schema_categories(restored)
    diff=''.join(difflib.unified_diff(raw_snapshot.splitlines(True),raw_restore.splitlines(True),fromfile='snapshot-schema.sql',tofile='restore-schema.sql'))
    item={'label':label,'snapshot':raw_snapshot,'restore':raw_restore,'diffComplete':diff,
      'categories':{key:{'snapshot':left[key],'restore':right[key],'equal':left[key]==right[key]} for key in left},
      'equal':snapshot==restored,'E5ProvenChecks':accepted,
      'allowedExceptions':['E1 owners','E2 ACL','E3 default ACL','E4 roles only in source','E5 associative AND with identical terms and PostgreSQL truth table'],
      'normalization':['CRLF to LF','trailing space','random restrict/unrestrict framing','non-SQL source/client version header; versions recorded separately','proven E5 only']}
    REPORT.setdefault('schemaComparisons',[]).append(item);save()
    return item['equal']

def rollback_drills(restored, baseline, cipher, phrase):
    if MODE=='local':
        local_sql(restored,'CREATE DATABASE p2_negative TEMPLATE postgres;')
        local_sql(restored,'ALTER TABLE public."BusinessScheduleWindow" DROP CONSTRAINT "BusinessScheduleWindow_minutes_check", ADD CONSTRAINT "BusinessScheduleWindow_minutes_check" CHECK (("startMinute" >= 0) AND ("startMinute" <= 1439) AND ("endMinute" >= 1) AND ("endMinute" <= 1440) AND ("startMinute" <= "endMinute"));',database='p2_negative')
        negative=local_schema(restored,'p2_negative')
        if schema_compare(baseline['schemaNormalized'],negative,'negative-actual-CHECK-changed',restored):
            raise RuntimeError('comparador-acepto-CHECK-alterado')
        old=check_lines(baseline['schemaNormalized'])['BusinessScheduleWindow_minutes_check'][1]
        new=check_lines(negative)['BusinessScheduleWindow_minutes_check'][1]
        if check_truth(restored,'BusinessScheduleWindow_minutes_check',old,new):
            raise RuntimeError('tabla-verdad-no-detecto-CHECK-alterado')
        check('control-negativo-CHECK-real-rechazado')
    local_sql(restored,'CREATE DATABASE p2_upgrade TEMPLATE postgres;')
    local_sql(restored,'REVOKE ALL ON DATABASE p2_upgrade FROM PUBLIC; GRANT CONNECT ON DATABASE p2_upgrade TO kortek_runtime,kortek_migrator;',database='p2_upgrade')
    migration=ROOT/'apps/api/prisma/migrations/20261003120000_customer_stage_one/migration.sql'
    checksum=hashlib.sha256(migration.read_bytes()).hexdigest()
    if checksum!='6e1d854f285f4a4a691377d6654d0a4df80f29ca5c3ad375b352ae96894d6dee':
        raise RuntimeError('migracion-28-cambio')
    local_sql(restored,migration.read_text(encoding='utf-8'),database='p2_upgrade',role='kortek_migrator')
    local_sql(restored,expand_sql(ROOT/'apps/api/ops/apply-runtime-grants.sql'),database='p2_upgrade')
    local_sql(restored,expand_sql(ROOT/'apps/api/ops/verify-runtime-role.sql'),database='p2_upgrade',role='kortek_runtime')
    after=local_census(restored,database='p2_upgrade',catalog=baseline['catalog'])
    if after['tables']!=baseline['tables'] or after['ledger']!=baseline['ledger']:
        REPORT.setdefault('rollbackDifferences',[]).append({'phase':'upgrade-preservation','snapshotTables':baseline['tables'],'observedTables':after['tables'],'snapshotLedger':baseline['ledger'],'observedLedger':after['ledger']});save()
        raise RuntimeError('upgrade-aislado-altero-datos-previos')
    api_rollback(restored)
    # El SQL local no simula registros Prisma aplicados: el ledger de recuperación queda 27.
    local_sql(restored,'BEGIN; CREATE TABLE public."P2RollbackFault"(id integer PRIMARY KEY); INSERT INTO public."P2RollbackFault" VALUES(1); INSERT INTO public."P2RollbackFault" VALUES(1); COMMIT;',
              database='p2_upgrade',expected=3)
    if local_sql(restored,"SELECT to_regclass('public.\"P2RollbackFault\"') IS NULL;",database='p2_upgrade')!='t':
        raise RuntimeError('fallo-transaccional-no-revertido')
    check('rollback-fallo-constraint-transaccion-aislada-sin-cambio-ledger')
    # La inversa solo se ensaya en este clon, nunca como rollback operativo QA.
    inverse=ROOT/'apps/api/prisma/rollback/customer-stage-one.disposable.sql'
    local_sql(restored,inverse.read_text(encoding='utf-8'),database='p2_upgrade')
    reverted=local_census(restored,database='p2_upgrade')
    for key in ['tables','ledger','counts']:
        if reverted[key]!=baseline[key]:
            REPORT.setdefault('rollbackDifferences',[]).append({'key':key,'snapshot':baseline[key],'restore':reverted[key]});save()
            raise RuntimeError('inversa-desechable-no-coincide')
    if not schema_compare(baseline['schemaNormalized'],local_schema(restored,'p2_upgrade'),'inversa-schema27',restored):
        raise RuntimeError('inversa-schema-difiere')
    check('inversa-desechable-reproduce-schema27-y-datos',operationalQaUse=False)
    # Ensayo de detección de corrupción de catálogo y recuperación en contenedor nuevo.
    local_sql(restored,'CREATE DATABASE p2_corrupt TEMPLATE postgres;')
    trigger=baseline['catalog']['triggers'][0] if baseline['catalog']['triggers'] else None
    if trigger:
        fault='ALTER TABLE public.'+ident(trigger['table_name'])+' DISABLE TRIGGER '+ident(trigger['name'])+';'
    else:
        constraint=baseline['catalog']['constraints'][0]
        fault='ALTER TABLE public.'+ident(constraint['table_name'])+' DROP CONSTRAINT '+ident(constraint['name'])+';'
    local_sql(restored,fault,database='p2_corrupt')
    corrupted=json.loads(local_sql(restored,CATALOG_SQL,database='p2_corrupt'))
    if corrupted==baseline['catalog']: raise RuntimeError('corrupcion-no-detectada')
    recovery=start('kortek-m2-c3-p2-recovery-'+RUN,server=True)
    restore_cipher(recovery,cipher,phrase,baseline)
    check('corrupcion-local-detectada-y-restaurada-en-contenedor-nuevo')
    REPORT['rollbackCases']=[
        {'case':'backup-restore-censo','response':'P2 sin congelamiento: parar ejecutor, conservar evidencia, no tocar servicios ni schema','tested':'restore aislado; P2 no controla servicios'},
        {'case':'migracion-lock-constraint','response':'mantener parada si ledger necesita conciliacion; nunca resolve automatico','tested':'fallo transaccional de constraint en clon, ledger sin cambio; sin Prisma fallido real'},
        {'case':'API-schema28-integro','response':'revertir solo codigo conservando schema y datos','tested':'artefacto anterior real sobre schema QA27->28, fixtures, recibos y vinculos preservados en ensayo previo'},
        {'case':'web','response':'alias QA al Preview anterior','tested':'solo baseline y decision documentada; no cambio de alias'},
        {'case':'Clerk','response':'propietario repone opciones anteriores','tested':'no ejecutado; prohibidos cambios Clerk'},
        {'case':'corrupcion','response':'copia cifrada del fallo y restore autorizado en destino nuevo; conciliar deltas','tested':'corrupcion de trigger en clon y restore de backup cifrado en contenedor nuevo; sin corrupcion/deltas QA reales'}]

def verify_writers(reader):
    # Lectura agregada: nunca query, conexiones, credenciales ni filas de negocio.
    # La falta de visibilidad sobre otras sesiones es condicion de parada.
    sql="""SELECT json_build_object(
      'roles',(SELECT json_agg(x ORDER BY x.role) FROM
        (SELECT rolname AS role,rolcanlogin AS login,rolsuper AS super,rolinherit AS inherit,
          rolcreatedb AS \"createDb\",rolbypassrls AS \"bypassRls\",rolcreaterole AS \"createRole\",
          rolreplication AS replication FROM pg_roles
          WHERE rolname IN ('kortek_migrator','kortek_runtime')) x),
      'otherRoleDml',(SELECT json_agg(x ORDER BY x.role) FROM
        (SELECT r.rolname AS role,count(*) FILTER (WHERE
          has_table_privilege(r.oid,c.oid,'INSERT,UPDATE,DELETE,TRUNCATE')) AS \"writeTables\"
          FROM pg_roles r CROSS JOIN pg_class c
          WHERE r.rolname IN ('anon','authenticated','authenticator','pgbouncer','service_role')
          AND c.relnamespace='public'::regnamespace AND c.relkind IN ('r','p') GROUP BY r.rolname) x),
      'membershipCount',(SELECT count(*) FROM pg_auth_members WHERE member IN
        (SELECT oid FROM pg_roles WHERE rolname IN ('kortek_runtime','kortek_migrator'))),
      'pgCronInstalled',EXISTS(SELECT 1 FROM pg_extension WHERE extname='pg_cron'),
      'preparedTransactions',(SELECT count(*) FROM pg_prepared_xacts WHERE database=current_database()),
      'uncontrolledWriters',(SELECT count(*) FROM pg_stat_activity WHERE datname=current_database()
        AND backend_type='client backend' AND pid<>pg_backend_pid() AND usename NOT IN
        ('kortek_runtime','authenticator','pgbouncer','supabase_admin','postgres')),
      'invisibleSessions',(SELECT count(*) FROM pg_stat_activity WHERE datname=current_database()
        AND backend_type='client backend' AND pid<>pg_backend_pid() AND state IS NULL),
      'activeClientTransactions',(SELECT count(*) FROM pg_stat_activity WHERE datname=current_database()
        AND backend_type='client backend' AND pid<>pg_backend_pid() AND xact_start IS NOT NULL)
    )"""
    observed=reader.json(sql,label='escritores-RO-agregado')
    prior=json.loads((ROOT/'docs/quality/evidence/m2-c3-p2/p2-segundo-intento.json').read_text(encoding='utf-8'))['writers']
    stable=['roles','otherRoleDml','membershipCount','pgCronInstalled']
    equal={key:observed[key]==prior[key] for key in stable}
    REPORT.setdefault('writersComparisons',[]).append({'baseline':{key:prior[key] for key in stable},
        'observed':observed,'equal':equal})
    save()
    if not all(equal.values()) or any(observed[key] for key in
        ['preparedTransactions','uncontrolledWriters','invisibleSessions']):
        raise RuntimeError('escritores-difieren-P0-o-visibilidad-incompleta')
    return observed

def cleanup():

def continue_existing_backup(prior):

def main():
