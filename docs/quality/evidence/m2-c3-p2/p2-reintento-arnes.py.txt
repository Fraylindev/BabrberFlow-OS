def old_ops(m,path):

def comparator():

    rows=json.loads((ROOT/'docs/quality/evidence/m2-c3-p2/p2-segundo-intento.json').read_text(encoding='utf-8'))['precheck']['ledger']
    assert len(rows)==31 and sum(r['rolledBack'] for r in rows)==4
    assert m.ledger_equal(rows,list(reversed(rows)))
    assert not m.ledger_equal(rows,rows[:-1])
    assert not m.ledger_equal(rows,rows+[rows[0]])
    mutated=[dict(r) for r in rows];mutated[0]['checksum']='0'*64
    assert not m.ledger_equal(rows,mutated)
    mutated=[dict(r) for r in rows];mutated[0]['finished']=not mutated[0]['finished']
    assert not m.ledger_equal(rows,mutated)
    REPORT['comparator']={'permutationAccepted':True,'multiplicityPreserved':True,'missingRejected':True,
        'extraRejected':True,'checksumRejected':True,'statusRejected':True}
