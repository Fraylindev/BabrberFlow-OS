def run(args, data=None, env=None, timeout=120, expected=0, label=None, record=True):
    if FREEZE_AT and not REPORT.get('qaResumed') and time.monotonic()-FREEZE_AT > 80*60 and not (label or '').startswith('reanudar-'):
        raise RuntimeError('ventana-limitada-80-minutos')
    p = subprocess.run(args,cwd=ROOT,input=data,capture_output=True,
                       env=env or BASE_ENV,timeout=timeout)
    if record:
        safe = ['--snapshot=<snapshot-en-memoria>' if x.startswith('--snapshot=') else x for x in args]
        REPORT['commands'].append({'args':safe,'exitCode':p.returncode,'label':label})
    if p.returncode != expected:
        REPORT['failureCommand']={'label':label,'exitCode':p.returncode}
        raise RuntimeError(label or 'comando-fallido')
    return p.stdout

def msys(path):

class Reader:
    def __init__(self, client):
        keys=['PGPASSWORD','PGHOST','PGPORT','PGUSER','PGDATABASE','PGSSLMODE','PGSSLROOTCERT','PGOPTIONS','PGAPPNAME']
        opts=sum((['-e',k] for k in keys),[])
        args=DOCKER+['exec','-i']+opts+[client,'psql','-X','-qAt','-v','ON_ERROR_STOP=1']
        self.proc=subprocess.Popen(args,cwd=ROOT,env=PG_ENV,stdin=subprocess.PIPE,
                                   stdout=subprocess.PIPE,stderr=subprocess.DEVNULL,text=True)
        self.lines=queue.Queue()
        def pump():
            for line in self.proc.stdout: self.lines.put(line.rstrip('\r\n'))
            self.lines.put(None)
        threading.Thread(target=pump,daemon=True).start()
        REPORT['commands'].append({'args':args,'label':'reader-QA-RR-RO'})
        self.q("BEGIN TRANSACTION ISOLATION LEVEL REPEATABLE READ READ ONLY; SET LOCAL statement_timeout='120s'")
    def q(self, sql):
        end='P2_END_'+secrets.token_hex(8)
        self.proc.stdin.write(sql+";\nSELECT '"+end+"';\n"); self.proc.stdin.flush()
        rows=[]
        while True:
            line=self.lines.get(timeout=150)
            if line is None: raise RuntimeError('reader-QA-fallo')
            if line==end: return rows
            if line: rows.append(line)
    def json(self, sql): return json.loads(self.q(sql)[0])
    def close(self):
        if self.proc.poll() is None:
            self.q('COMMIT'); self.proc.stdin.close()
            if self.proc.wait(timeout=15)!=0: raise RuntimeError('reader-QA-exit')
    def kill(self):
        if self.proc.poll() is None: self.proc.kill(); self.proc.wait(timeout=15)

def resume():
    with RESUME_LOCK:
        if FREEZE_AT is None or REPORT.get('qaResumed'): return
        # Siempre recuperar API antes del worker; no modificar imagen/configuración.
        failures=[]
        for unit in ['kortek-api-staging','kortek-email-worker-staging']:
            try:
                ssh('sudo -n systemctl start '+unit+'; systemctl is-active '+unit,label='reanudar-'+unit)
            except Exception:
                failures.append(unit)
        if failures:
            REPORT['resumeFailures']=failures; save()
            raise RuntimeError('reanudacion-QA-no-completa')
        REPORT['qaResumed']=True
        REPORT['outageSeconds']=round(time.monotonic()-FREEZE_AT,3)
        REPORT['resumeOrder']=['kortek-api-staging','kortek-email-worker-staging']
        save()

def watchdog():
    REPORT['watchdogTriggered']=True
    try: resume()
    finally:
        erase(PHRASE); erase(DECRYPT_PHRASE)
        if SOURCE:
            try: SOURCE.kill()
            except Exception: pass
        if PG_ENV: PG_ENV.clear()
        try: cleanup()
        except Exception: REPORT['cleanupFailure']=True
        REPORT['success']=False
        REPORT['failure']={'stage':STAGE,'label':'watchdog-85-minutos'}
        save()
        os._exit(2)

def request_phrase(kind):
    emit('prompt-frase',kind=kind)
    line=sys.stdin.buffer.readline()
    if not line: raise RuntimeError('prompt-cerrado')
    value=bytearray(line.rstrip(b'\r\n')); line=None
    if not value: raise RuntimeError('frase-vacia')
    return value

def gpg_args(phrase, path, encrypt=True):
    home=WORK/'gpg-home'; home.mkdir(exist_ok=True)
    args=[GPG,'--no-options','--homedir',msys(home),'--no-symkey-cache','--batch',
          '--pinentry-mode','loopback','--passphrase-fd','0']
    return args+(['--symmetric','--cipher-algo','AES256','--output',msys(path)] if encrypt else ['--decrypt',msys(path)])

def encrypt_data(phrase, data, dest):
    # GPG consume la primera línea del stdin como frase; el resto es el payload.
    return run(gpg_args(phrase,dest),data=bytes(phrase)+b'\n'+data,
               label='cifrado-AES256-stdin-anonimo')

def decrypt_data(phrase, path):
    return run(gpg_args(phrase,path,False),data=bytes(phrase)+b'\n',label='descifrado-stdin-anonimo')

def crypto_probe():
    key=bytearray(secrets.token_urlsafe(24).encode())
    path=WORK/'prueba-sintetica.gpg'
    try:
        encrypt_data(key,b'P2 LOCAL SIN DATOS DE NEGOCIO\n',path)
        if decrypt_data(key,path)!=b'P2 LOCAL SIN DATOS DE NEGOCIO\n':
            raise RuntimeError('cifrado-pipeline-invalido')
        check('pipeline-stdin-frase-y-datos-validado')
    finally:
        erase(key)
        if path.exists(): path.unlink()

def census(query, catalog=None):
    catalog=catalog or json.loads(query(CATALOG_SQL))
    values={}
    for table in catalog['tables']:
        name=table['name']
        cols=[c['name'] for c in catalog['columns'] if c['table_name']==name]
        select=','.join(ident(c) for c in cols)
        sql="SELECT json_build_object('count',count(*),'fingerprint',coalesce(md5(string_agg(md5(to_jsonb(t)::text),'' ORDER BY md5(to_jsonb(t)::text) COLLATE \"C\")),md5(''))) FROM (SELECT "+select+" FROM public."+ident(name)+") t"
        values[name]=json.loads(query(sql))
    return {'catalog':catalog,'tables':values,'ledger':json.loads(query(LEDGER_SQL)),
            'counts':json.loads(query(COUNTS_SQL)),'roles':json.loads(query(ROLE_SQL))}

def roles_prepare(container, catalog, roles, database='postgres'):
    if roles['memberships']: raise RuntimeError('membresias-de-interes-no-vacias')
    owners={x['owner'] for kind in ['tables','types','functions'] for x in catalog[kind]}
    owners.add(catalog['schemaOwner'])
    attrs={x['rolname']:x for x in roles['attributes']}
    for name in sorted(owners|set(attrs)):
        if name=='postgres': continue
        flags=attrs.get(name)
        if flags:
            options=' '.join(yes if flags[key] else no for key,yes,no in [
                ('rolcanlogin','LOGIN','NOLOGIN'),('rolinherit','INHERIT','NOINHERIT'),
                ('rolsuper','SUPERUSER','NOSUPERUSER'),('rolcreatedb','CREATEDB','NOCREATEDB'),
                ('rolcreaterole','CREATEROLE','NOCREATEROLE'),('rolreplication','REPLICATION','NOREPLICATION'),
                ('rolbypassrls','BYPASSRLS','NOBYPASSRLS')])
        else: options='NOLOGIN NOSUPERUSER NOCREATEDB NOCREATEROLE NOREPLICATION NOBYPASSRLS NOINHERIT'
        local_sql(container,'CREATE ROLE '+ident(name)+' '+options+';',database)

def roles_apply(container, catalog, roles, database='postgres'):
    commands=['ALTER SCHEMA public OWNER TO '+ident(catalog['schemaOwner'])+';',
              'REVOKE ALL ON SCHEMA public FROM PUBLIC,kortek_runtime,kortek_migrator,kortek_backup;',
              'REVOKE ALL ON ALL TABLES IN SCHEMA public FROM PUBLIC,kortek_runtime,kortek_migrator,kortek_backup;']
    for table in catalog['tables']:
        commands.append('ALTER TABLE public.'+ident(table['name'])+' OWNER TO '+ident(table['owner'])+';')
    for t in catalog['types']:
        commands.append('ALTER TYPE public.'+ident(t['name'])+' OWNER TO '+ident(t['owner'])+';')
    for f in catalog['functions']:
        commands+=['ALTER FUNCTION '+f['signature']+' OWNER TO '+ident(f['owner'])+';',
                   'REVOKE ALL ON FUNCTION '+f['signature']+' FROM PUBLIC,kortek_runtime,kortek_migrator,kortek_backup;']
    for p in roles['tablePrivileges']:
        if p['allowed']: commands.append('GRANT '+p['privilege']+' ON TABLE public.'+ident(p['table_name'])+' TO '+ident(p['role'])+';')
    for p in roles['schemaPrivileges']:
        if p['allowed']: commands.append('GRANT '+p['privilege']+' ON SCHEMA public TO '+ident(p['role'])+';')
    for p in roles['functionPrivileges']:
        if p['allowed']: commands.append('GRANT EXECUTE ON FUNCTION '+p['signature']+' TO '+ident(p['role'])+';')
    kinds={'r':'TABLES','S':'SEQUENCES','f':'FUNCTIONS','T':'TYPES','n':'SCHEMAS'}
    for p in roles['defaults']:
        prefix='ALTER DEFAULT PRIVILEGES FOR ROLE '+ident(p['owner'])
        if p['namespace']: prefix+=' IN SCHEMA '+ident(p['namespace'])
        target='PUBLIC' if p['grantee']=='PUBLIC' else ident(p['grantee'])
        commands.append(prefix+' GRANT '+p['privilege']+' ON '+kinds[p['kind']]+' TO '+target+(' WITH GRANT OPTION' if p['grantable'] else '')+';')
    local_sql(container,'BEGIN;\n'+'\n'.join(commands)+'\nCOMMIT;',database)

def filter_toc(toc):
    selected=[]; removed=[]
    for line in toc.splitlines():
        if line.startswith(';') or not line.strip(): selected.append(line); continue
        if re.search(r'\bSCHEMA - public\b|\bEXTENSION - btree_gist\b|\bCOMMENT - EXTENSION btree_gist\b|\bCOMMENT - SCHEMA public\b',line):
            removed.append(line)
        else: selected.append(line)
    if any('TABLE DATA' in x for x in removed): raise RuntimeError('TOC-omite-tablas')
    REPORT['tocBootstrapEntriesExcluded']=len(removed)
    return ('\n'.join(selected)+'\n').encode()

def restore_bytes(container, payload, database='postgres'):
    local_sql(container,'CREATE EXTENSION IF NOT EXISTS btree_gist WITH SCHEMA public;',database)
    toc=docker('exec','-i',container,'pg_restore','--list',data=payload,label='TOC-schema-local').decode()

def stream_encrypt(client, snapshot, phrase, dest):
    args=dump_args(client,snapshot)
    src=subprocess.Popen(args,cwd=ROOT,env=PG_ENV,stdout=subprocess.PIPE,stderr=subprocess.DEVNULL)
    enc_args=gpg_args(phrase,dest)
    enc=subprocess.Popen(enc_args,cwd=ROOT,env=BASE_ENV,stdin=subprocess.PIPE,stdout=subprocess.DEVNULL,stderr=subprocess.DEVNULL)
    try:
        enc.stdin.write(bytes(phrase)+b'\n'); enc.stdin.flush()
        shutil.copyfileobj(src.stdout,enc.stdin,1024*1024)
        enc.stdin.close(); src.stdout.close()
        src_code=src.wait(timeout=150); enc_code=enc.wait(timeout=150)
        REPORT['commands'].append({'args':['--snapshot=<snapshot-en-memoria>' if x.startswith('--snapshot=') else x for x in args],
                                   'exitCode':src_code,'pipeArgs':enc_args,'pipeExitCode':enc_code})
        if src_code or enc_code: raise RuntimeError('dump-o-cifrado-fallo')
    finally:
        for p in [src,enc]:
            if p.poll() is None: p.kill(); p.wait(timeout=15)

def stream_decrypt_restore(container, phrase, path, database='postgres'):
    args=gpg_args(phrase,path,False)
    dec=subprocess.Popen(args,cwd=ROOT,env=BASE_ENV,stdin=subprocess.PIPE,stdout=subprocess.PIPE,stderr=subprocess.DEVNULL)
    restore_args=DOCKER+['exec','-i',container,'pg_restore','-U','postgres','-d',database,
                        '--single-transaction','--exit-on-error','--no-owner','--no-privileges','--use-list=/tmp/p2-restore.list']
    proc=subprocess.Popen(restore_args,cwd=ROOT,env=BASE_ENV,stdin=dec.stdout,
                          stdout=subprocess.DEVNULL,stderr=subprocess.DEVNULL)
    dec.stdout.close()
    try:
        dec.stdin.write(bytes(phrase)+b'\n'); dec.stdin.close()
        b=proc.wait(timeout=300); a=dec.wait(timeout=30)
        REPORT['commands'].append({'args':args,'exitCode':a,'pipeArgs':restore_args,'pipeExitCode':b})
        if a or b: raise RuntimeError('descifrado-restore-fallo')
    finally:
        for p in [dec,proc]:
            if p.poll() is None: p.kill(); p.wait(timeout=15)

def decrypt_toc(container, phrase, path):
    args=gpg_args(phrase,path,False)
    dec=subprocess.Popen(args,cwd=ROOT,env=BASE_ENV,stdin=subprocess.PIPE,
                         stdout=subprocess.PIPE,stderr=subprocess.DEVNULL)
    proc=subprocess.Popen(DOCKER+['exec','-i',container,'pg_restore','--list'],
                          cwd=ROOT,env=BASE_ENV,stdin=dec.stdout,stdout=subprocess.PIPE,stderr=subprocess.DEVNULL)
    dec.stdout.close()
    try:
        dec.stdin.write(bytes(phrase)+b'\n'); dec.stdin.close()
        listing=proc.communicate(timeout=180)[0]; a=dec.wait(timeout=30)
        REPORT['commands'].append({'args':args,'exitCode':a,'pipeTo':'pg_restore --list','pipeExitCode':proc.returncode})
        if a or proc.returncode: raise RuntimeError('descifrado-TOC-fallo')
        docker('exec','-i',container,'sh','-c','cat > /tmp/p2-restore.list',
               data=filter_toc(listing.decode()),label='TOC-public-btree-tmpfs')
    finally:
        for p in [dec,proc]:
            if p.poll() is None: p.kill(); p.wait(timeout=15)

def restore_cipher(container, cipher, phrase, baseline):
    roles_prepare(container,baseline['catalog'],baseline['roles'])
    local_sql(container,'CREATE EXTENSION btree_gist WITH SCHEMA public;')
    decrypt_toc(container,phrase,cipher)
    stream_decrypt_restore(container,phrase,cipher)
    roles_apply(container,baseline['catalog'],baseline['roles'])
    result=census(lambda sql:local_sql(container,sql))
    equal={k:result[k]==baseline[k] for k in ['catalog','tables','ledger','counts','roles']}
    check('cotejo-restore-snapshot',**equal,tableCount=len(result['tables']))
    if not all(equal.values()): raise RuntimeError('restore-desigualdad')
    return result

def expand_sql(path):
    text=path.read_text()
    return re.sub(r'^\\ir ([^\r\n]+)$',lambda m:expand_sql(path.parent/m[1].strip()),text,flags=re.M)

def rollback_drills(restored, baseline, cipher, phrase):
    local_sql(restored,'CREATE DATABASE p2_upgrade TEMPLATE postgres;')
    migration=ROOT/'apps/api/prisma/migrations/20261003120000_customer_stage_one/migration.sql'
    checksum=hashlib.sha256(migration.read_bytes()).hexdigest()
    if checksum!='6e1d854f285f4a4a691377d6654d0a4df80f29ca5c3ad375b352ae96894d6dee':
        raise RuntimeError('migracion-28-cambio')
    local_sql(restored,migration.read_text(),database='p2_upgrade',role='kortek_migrator')
    local_sql(restored,expand_sql(ROOT/'apps/api/ops/apply-runtime-grants.sql'),database='p2_upgrade')
    local_sql(restored,expand_sql(ROOT/'apps/api/ops/verify-runtime-role.sql'),database='p2_upgrade',role='kortek_runtime')
    after=census(lambda sql:local_sql(restored,sql,database='p2_upgrade'),catalog=baseline['catalog'])
    if after['tables']!=baseline['tables'] or after['ledger']!=baseline['ledger']:
        raise RuntimeError('upgrade-aislado-altero-datos-previos')
    # El SQL local no simula registros Prisma aplicados: el ledger de recuperación queda 27.
    local_sql(restored,'BEGIN; CREATE TABLE public."P2RollbackFault"(id integer PRIMARY KEY); INSERT INTO public."P2RollbackFault" VALUES(1); INSERT INTO public."P2RollbackFault" VALUES(1); COMMIT;',
              database='p2_upgrade',expected=1)
    if local_sql(restored,"SELECT to_regclass('public.\"P2RollbackFault\"') IS NULL;",database='p2_upgrade')!='t':
        raise RuntimeError('fallo-transaccional-no-revertido')
    check('rollback-fallo-constraint-transaccion-aislada-sin-cambio-ledger')
    # La inversa solo se ensaya en este clon, nunca como rollback operativo QA.
    inverse=ROOT/'apps/api/prisma/rollback/customer-stage-one.disposable.sql'
    local_sql(restored,inverse.read_text(),database='p2_upgrade')
    reverted=census(lambda sql:local_sql(restored,sql,database='p2_upgrade'))
    for key in ['catalog','tables','ledger','counts']:
        if reverted[key]!=baseline[key]: raise RuntimeError('inversa-desechable-no-coincide')
    check('inversa-desechable-reproduce-schema27-y-datos',operationalQaUse=False)
    # Ensayo de detección de corrupción de catálogo y recuperación en contenedor nuevo.
    local_sql(restored,'CREATE DATABASE p2_corrupt TEMPLATE postgres;')
    trigger=baseline['catalog']['triggers'][0] if baseline['catalog']['triggers'] else None
    if trigger:
        fault='ALTER TABLE public.'+ident(trigger['table_name'])+' DISABLE TRIGGER '+ident(trigger['name'])+';'
    else:
        constraint=baseline['catalog']['constraints'][0]
        fault='ALTER TABLE public.'+ident(constraint['table_name'])+' DROP CONSTRAINT '+ident(constraint['name'])+';'
    local_sql(restored,fault,database='p2_corrupt')
    corrupted=json.loads(local_sql(restored,CATALOG_SQL,database='p2_corrupt'))
    if corrupted==baseline['catalog']: raise RuntimeError('corrupcion-no-detectada')
    recovery=start('kortek-m2-c3-p2-recovery-'+RUN,server=True)
    restore_cipher(recovery,cipher,phrase,baseline)
    check('corrupcion-local-detectada-y-restaurada-en-contenedor-nuevo')
    REPORT['rollbackCases']=[
        {'case':'backup-restore-censo','response':'reanudar QA previa sin tocar schema','tested':'reanudacion real al cierre y restore aislado'},
        {'case':'migracion-lock-constraint','response':'mantener parada si ledger necesita conciliacion; nunca resolve automatico','tested':'fallo transaccional de constraint en clon, ledger sin cambio; sin Prisma fallido real'},
        {'case':'API-schema28-integro','response':'revertir solo codigo conservando schema y datos','tested':'artefacto anterior real sobre schema QA27->28, fixtures, recibos y vinculos preservados en ensayo previo'},
        {'case':'web','response':'alias QA al Preview anterior','tested':'solo baseline y decision documentada; no cambio de alias'},
        {'case':'Clerk','response':'propietario repone opciones anteriores','tested':'no ejecutado; prohibidos cambios Clerk'},
        {'case':'corrupcion','response':'copia cifrada del fallo y restore autorizado en destino nuevo; conciliar deltas','tested':'corrupcion de trigger en clon y restore de backup cifrado en contenedor nuevo; sin corrupcion/deltas QA reales'}]

        if baseline['ledger']!=prior['precheck']['ledger']: raise RuntimeError('ledger-snapshot-difiere')
        custody=Path('C:/KortekBackups/qa-m2-c3')
        basename='qa-m2-c3-p2-'+datetime.now(timezone.utc).strftime('%Y%m%dT%H%M%SZ')+'-'+RUN
        cipher=custody/(basename+'.dump.gpg')
        manifest=custody/(basename+'.censo.gpg')
        if cipher.exists() or manifest.exists(): raise RuntimeError('custodia-colision')
        REPORT['backupFile']=str(cipher); REPORT['censusEncryptedFile']=str(manifest)
        stream_encrypt(client,snapshot,PHRASE,cipher)
        encrypt_data(PHRASE,json.dumps(baseline,ensure_ascii=True).encode(),manifest)
        erase(PHRASE); PHRASE=None
        SOURCE.close(); SOURCE=None
        REPORT['backupSha256']=hashlib.sha256(cipher.read_bytes()).hexdigest()
        REPORT['censusEncryptedSha256']=hashlib.sha256(manifest.read_bytes()).hexdigest()
        REPORT['backupBytes']=cipher.stat().st_size
        REPORT['snapshotCensus']={name:value['count'] for name,value in baseline['tables'].items()}
        REPORT['snapshotCounts']=baseline['counts']
        REPORT['snapshotLedgerApplied']=27
        check('respaldo-cifrado-y-censo-misma-snapshot',backupBytes=REPORT['backupBytes'])
        DECRYPT_PHRASE=request_phrase('descifrar-reintroduccion-independiente')
        decoded=json.loads(decrypt_data(DECRYPT_PHRASE,manifest))
        if decoded!=baseline: raise RuntimeError('censo-cifrado-no-coincide')
        STAGE='restore-cotejo-rollback'
        restored=start('kortek-m2-c3-p2-restore-'+RUN,server=True)
        restore_cipher(restored,cipher,DECRYPT_PHRASE,baseline)
        rollback_drills(restored,baseline,cipher,DECRYPT_PHRASE)
        erase(DECRYPT_PHRASE); DECRYPT_PHRASE=None
        REPORT['backupAccepted']=True
        REPORT['success']=True
    except Exception as error:
        REPORT['success']=False
        REPORT['failure']={'stage':STAGE,'errorClass':type(error).__name__,
                           'label':str(error) if isinstance(error,RuntimeError) else 'detalle-omitido'}
        emit('parada',**REPORT['failure'])
    finally:
        erase(PHRASE); erase(DECRYPT_PHRASE); PHRASE=None; DECRYPT_PHRASE=None
        if SOURCE:
            try: SOURCE.close()
            except Exception: SOURCE.kill()
            SOURCE=None
        if PG_ENV: PG_ENV['PGPASSWORD']=''; PG_ENV.clear(); PG_ENV=None
        try:
            resume()
            REPORT['finalInfra']=infra()
            REPORT['finalEffective']=effective()
            if FREEZE_AT is not None:
                REPORT['publicBooking']=public_get()
                REPORT['p3bRequiresNewBackupIfAnyWrite']=True
                REPORT['publicGetMayWriteRateBuckets']=True
        except Exception as error:
            REPORT['success']=False
            REPORT['recoveryFailure']={'errorClass':type(error).__name__}
        if WATCHDOG: WATCHDOG.cancel()
        try: cleanup()
        except Exception: REPORT['success']=False; REPORT['cleanupFailure']=True
        REPORT['finishedUtc']=datetime.now(timezone.utc).isoformat()
        save(); emit('terminado',success=REPORT.get('success',False),report=str(WORK/'result.json'))

if __name__=='__main__':
    main()
    sys.exit(0 if REPORT.get('success') else 1)
