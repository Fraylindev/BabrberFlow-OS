def emit(event, **fields):
    print(json.dumps({'stage':event, **fields}),flush=True)

def mark(label, **fields):
    REPORT['checks'].append({'label':label, **fields}); emit(label, **fields)

def run(args, input=None, env=None, timeout=120, expected=0, label=None, record=True):
    start=time.monotonic()
    p=subprocess.run(args,cwd=ROOT,input=input,capture_output=True,env=env or BASE_ENV,timeout=timeout)
    if record:
        REPORT['commands'].append({'command':args,'exitCode':p.returncode,'durationSeconds':round(time.monotonic()-start,3),'label':label})
    if p.returncode != expected:
        raise RuntimeError(label or 'command-failed')
    return p.stdout

def docker(*args, **kwargs): return run(DOCKER+list(args), **kwargs)

class Reader:
    def __init__(self, client):
        env_args=[]
        for k in ['PGPASSWORD','PGHOST','PGPORT','PGUSER','PGDATABASE','PGSSLMODE','PGSSLROOTCERT','PGOPTIONS','PGAPPNAME']: env_args+=['-e',k]
        self.proc=subprocess.Popen(DOCKER+['exec','-i']+env_args+[client,'psql','-X','-qAt','-v','ON_ERROR_STOP=1'],cwd=ROOT,env=PG_ENV,stdin=subprocess.PIPE,stdout=subprocess.PIPE,stderr=subprocess.PIPE,text=True)
        self.lines=queue.Queue()
        def pump():
            for line in self.proc.stdout: self.lines.put(line.rstrip('\r\n'))
            self.lines.put(None)
        threading.Thread(target=pump,daemon=True).start()
        threading.Thread(target=lambda:self.proc.stderr.read(),daemon=True).start()
    def q(self, sql):
        token='P2_END_'+secrets.token_hex(8)
        self.proc.stdin.write(sql+';\nSELECT \''+token+'\';\n'); self.proc.stdin.flush()
        lines=[]
        while True:
            line=self.lines.get(timeout=150)
            if line is None: raise RuntimeError('source-reader-failed')
            if line==token: return lines
            if line: lines.append(line)
    def close(self, commit=True):
        if self.proc.poll() is None:
            if commit: self.q('COMMIT')
            self.proc.stdin.close()
            code=self.proc.wait(timeout=15)
            if code!=0: raise RuntimeError('source-reader-exit')
    def kill(self):
        if self.proc.poll() is None: self.proc.kill(); self.proc.wait(timeout=15)

def filter_toc(toc):
    lines=[]; excluded=[]
    for line in toc.splitlines():
        if line.startswith(';') or not line.strip(): lines.append(line); continue
        if re.search(r'\bSCHEMA - public\b|\bEXTENSION - btree_gist\b|\bCOMMENT - EXTENSION btree_gist\b|\bCOMMENT - SCHEMA public\b',line): excluded.append(line)
        else: lines.append(line)
    assert all('TABLE DATA' not in x for x in excluded)
    REPORT['tocExcluded']=excluded
    return '\n'.join(lines)+'\n'

finally:
    if SOURCE: SOURCE.kill()
    for name in list(OWNED):
        try: remove(name)
        except Exception as error: REPORT['failures'].append({'stage':'cleanup','errorClass':type(error).__name__}); REPORT['success']=False
    (WORK/('last-'+sys.argv[1]+'.json')).write_text(json.dumps(REPORT,indent=2)+'\n')
sys.exit(0 if REPORT.get('success') else 1)
