const fs=require('node:fs'),path=require('node:path'),{spawnSync}=require('node:child_process');
const root=process.cwd(),dir=path.join(root,'.tmp/ci-deps-checks');fs.mkdirSync(dir,{recursive:true});
const env=Object.fromEntries(Object.entries(process.env).filter(([k])=>!/DATABASE|POSTGRES|^PG|CLERK|JWT|CLOUDINARY|RESEND|EMAIL|NOTIFICATION|TOKEN|SECRET|PASSWORD|NODE_OPTIONS/.test(k)));
env.NODE_OPTIONS='--require="'+path.join(root,'apps/api/test/guest-postgres-no-dotenv.cjs').replaceAll('\\','/')+'" --require="'+path.join(root,'apps/api/test/guest-network-guard.cjs').replaceAll('\\','/')+'"';
env.pnpm_config_verify_deps_before_run='false';env.NEXT_TELEMETRY_DISABLED='1';env.DEPLOY_ENV='development';env.NEXT_PUBLIC_API_URL='http://127.0.0.1:1';
env.NEXT_PUBLIC_CLERK_PUBLISHABLE_KEY='pk_test_'+Buffer.from('synthetic.clerk.accounts.dev$').toString('base64');env.CLERK_SECRET_KEY='sk_test_synthetic_local_only';
const cli=path.join(process.env.APPDATA,'npm/node_modules/pnpm/bin/pnpm.cjs');
const checks=[];
for(const [label,args] of [
 ['prisma-generate',['--filter','api','exec','prisma','generate']],
 ['api-types',['--filter','api','type-check']],
 ['web-types',['--filter','web','type-check:clean']],
 ['api-lint',['--filter','api','lint']],
 ['web-lint',['--filter','web','lint']],
 ['api-build',['--filter','api','build']],
 ['web-unit',['--filter','web','test:unit']],
 ['web-components',['--filter','web','exec','vitest','run','--config',path.join(root,'.tmp/retry-vitest.config.mts')]],
].filter(([label])=>!process.env.CI_DEPS_ONLY||label===process.env.CI_DEPS_ONLY)) {
 const r=spawnSync(process.execPath,[cli,...args],{cwd:root,env,encoding:'utf8',windowsHide:true,timeout:900000,maxBuffer:20e6});
 fs.writeFileSync(path.join(dir,label+'.log'),(r.stdout||'')+(r.stderr||'')+(r.error?.message||''));
 checks.push({label,args,exitCode:r.status});fs.writeFileSync(path.join(dir,process.env.CI_DEPS_ONLY?'retry-'+process.env.CI_DEPS_ONLY+'.json':'results.json'),JSON.stringify(checks,null,2)+'\n');console.log(label+': exit='+r.status);
}
process.exitCode=checks.every(c=>c.exitCode===0)?0:1;
